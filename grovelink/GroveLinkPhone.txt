// GroveLink Phone v5 — crash-safer. No #model names, no 033E GXT, no hold_cellphone.
// K toggle | UP/DOWN | ENTER or SPACE select | BACKSPACE close
// Menu labels via 0ACD highpriority (same pattern as existing prompts).
0000: NOP
03A4: name_thread 'GROVELNK'

20@ = 0
21@ = 0
22@ = 0
23@ = 0
24@ = 0
25@ = 0

:MAIN
0001: wait 0
00D6: if
0256:   player $PLAYER_CHAR defined
004D: jf @MAIN
00D6: if
8118:   not actor $PLAYER_ACTOR dead
004D: jf @FORCE_CLOSE

00D6: if
0AB0:   key_pressed 75
004D: jf @AFTER_TOGGLE
00D6: if
0039:   23@ == 0
004D: jf @AFTER_TOGGLE
0006: 23@ = 1
00D6: if
0039:   20@ == 0
004D: jf @PUT_AWAY
0006: 20@ = 1
0006: 21@ = 0
0006: 22@ = 0
0ACD: show_text_highpriority "CAMERA / INBOX / CLOSE" time 1800
018C: play_sound 1138 at 0.0 0.0 0.0
0002: jump @AFTER_TOGGLE

:PUT_AWAY
0006: 20@ = 0
0006: 21@ = 0
0ACD: show_text_highpriority "GROVELINK CLOSED" time 1200
018C: play_sound 1139 at 0.0 0.0 0.0
0002: jump @AFTER_TOGGLE

:FORCE_CLOSE
0006: 20@ = 0
0002: jump @MAIN

:AFTER_TOGGLE
00D6: if
8039:   not 23@ == 0
004D: jf @PHONE_LOGIC
00D6: if
8AB0:   not key_pressed 75
004D: jf @PHONE_LOGIC
0006: 23@ = 0

:PHONE_LOGIC
00D6: if
0039:   20@ == 1
004D: jf @MAIN
03F0: enable_text_draw 1
038E: draw_box 548.0 248.0 132.0 292.0 28 28 32 255
038E: draw_box 548.0 248.0 118.0 276.0 12 12 14 255
038E: draw_box 548.0 232.0 102.0 196.0 8 28 16 255
038E: draw_box 548.0 140.0 102.0 14.0 6 18 10 255
038E: draw_box 548.0 372.0 16.0 16.0 50 50 54 255

00D6: if
0AB0:   key_pressed 40
004D: jf @KEY_UP
0001: wait 160
000A: 22@ += 1
00D6: if
0019:   22@ > 2
004D: jf @SHOW_SEL
0006: 22@ = 0
0002: jump @SHOW_SEL
:KEY_UP
00D6: if
0AB0:   key_pressed 38
004D: jf @PHONE_LOGIC2
0001: wait 160
000E: 22@ -= 1
00D6: if
001B:   0 > 22@
004D: jf @SHOW_SEL
0006: 22@ = 2

:SHOW_SEL
00D6: if
0039:   22@ == 0
004D: jf @SHOW_SEL2
0ACD: show_text_highpriority "CAMERA  Enter/Space snap" time 1200
0002: jump @PHONE_LOGIC2
:SHOW_SEL2
00D6: if
0039:   22@ == 1
004D: jf @SHOW_SEL3
0ACD: show_text_highpriority "INBOX" time 1200
0002: jump @PHONE_LOGIC2
:SHOW_SEL3
0ACD: show_text_highpriority "CLOSE" time 1200

:PHONE_LOGIC2
// Enter (13) or Space (32) to activate
00D6: if or
0AB0:   key_pressed 13
0AB0:   key_pressed 32
004D: jf @KEY_BACK
0001: wait 180
00D6: if
0039:   22@ == 0
004D: jf @SEL2
0A2F: set_photo_camera_effect 1
0001: wait 80
0A1E: take_photo 1
0001: wait 200
0A2F: set_photo_camera_effect 0
0ACD: show_text_highpriority "PHOTO TAKEN" time 2000
00D6: if
0AAB:   file_exists "cleo\GroveLink\link.ini"
004D: jf @KEY_BACK
0AF1: write_int 1 to_ini_file "cleo\GroveLink\link.ini" section "PHOTO" key "take"
0AF0: 25@ = read_int_from_ini_file "cleo\GroveLink\link.ini" section "PHOTO" key "count"
000A: 25@ += 1
0AF1: write_int 25@ to_ini_file "cleo\GroveLink\link.ini" section "PHOTO" key "count"
0002: jump @KEY_BACK
:SEL2
00D6: if
0039:   22@ == 1
004D: jf @SEL3
0ACD: show_text_highpriority "INBOX  use real phone page" time 2000
0002: jump @KEY_BACK
:SEL3
0006: 20@ = 0
0ACD: show_text_highpriority "GROVELINK CLOSED" time 1200

:KEY_BACK
00D6: if
0AB0:   key_pressed 8
004D: jf @MAIN
0006: 20@ = 0
0001: wait 160
0002: jump @MAIN
