// GroveLink Phone v8 — crash-safer. No #model names, no 033E GXT, no hold_cellphone.
// K toggle | UP/DOWN wrap | ENTER or SPACE select | BACKSPACE close
// Menu: CAMERA / INBOX / CONTACTS / STATUS / HELP / CLOSE — labels via 0ACD only.
// INBOX plays 018C when new=1. CONTACTS cycles static flavor lines (no models).
// STATUS shows BRIDGE LIVE / NO BRIDGE + shot count from link.ini.
// HELP mentions UPDATE_GROVELINK for outdated installs.
0000: NOP
03A4: name_thread 'GROVELNK'

20@ = 0 // phone open
21@ = 0
22@ = 0 // menu index 0=CAMERA 1=INBOX 2=CONTACTS 3=STATUS 4=HELP 5=CLOSE
23@ = 0 // K debounce
24@ = 0 // temp int (INBOX.new / STATUS.bridge)
25@ = 0 // PHOTO.count
26@ = 0 // contacts cycle index 0=Sweet 1=Smoke 2=Ryder 3=Cesar
27@ = 0 // phone-just-opened edge (remind / setup once per open)

:MAIN
0001: wait 0
00D6: if
0256:   player $PLAYER_CHAR defined
004D: jf @MAIN
00D6: if
8118:   not actor $PLAYER_ACTOR dead
004D: jf @FORCE_CLOSE

00D6: if
0AB0:   key_pressed 75
004D: jf @AFTER_TOGGLE
00D6: if
0039:   23@ == 0
004D: jf @AFTER_TOGGLE
0006: 23@ = 1
00D6: if
0039:   20@ == 0
004D: jf @PUT_AWAY
0006: 20@ = 1
0006: 21@ = 0
0006: 22@ = 0
0006: 27@ = 1
0ACD: show_text_highpriority "CAM INBOX CONTACTS STATUS HELP CLOSE" time 1800
018C: play_sound 1138 at 0.0 0.0 0.0
0002: jump @AFTER_TOGGLE

:PUT_AWAY
0006: 20@ = 0
0006: 21@ = 0
0006: 27@ = 0
0ACD: show_text_highpriority "GROVELINK CLOSED" time 1200
018C: play_sound 1139 at 0.0 0.0 0.0
0002: jump @AFTER_TOGGLE

:FORCE_CLOSE
0006: 20@ = 0
0006: 27@ = 0
0002: jump @MAIN

:AFTER_TOGGLE
00D6: if
8039:   not 23@ == 0
004D: jf @PHONE_LOGIC
00D6: if
8AB0:   not key_pressed 75
004D: jf @PHONE_LOGIC
0006: 23@ = 0

:PHONE_LOGIC
00D6: if
0039:   20@ == 1
004D: jf @MAIN

// Once per open: if bridge is down, remind (STATUS.bridge=0). No wait spam.
00D6: if
0039:   27@ == 1
004D: jf @PHONE_DRAW
0006: 27@ = 0
00D6: if
0AAB:   file_exists "cleo\GroveLink\link.ini"
004D: jf @PHONE_DRAW
0AF0: 24@ = read_int_from_ini_file "cleo\GroveLink\link.ini" section "STATUS" key "bridge"
00D6: if
0039:   24@ == 0
004D: jf @PHONE_DRAW
0ACD: show_text_highpriority "START GROVELINK BRIDGE" time 2500

:PHONE_DRAW
03F0: enable_text_draw 1
038E: draw_box 548.0 248.0 132.0 292.0 28 28 32 255
038E: draw_box 548.0 248.0 118.0 276.0 12 12 14 255
038E: draw_box 548.0 232.0 102.0 196.0 8 28 16 255
038E: draw_box 548.0 140.0 102.0 14.0 6 18 10 255
038E: draw_box 548.0 372.0 16.0 16.0 50 50 54 255

00D6: if
0AB0:   key_pressed 40
004D: jf @KEY_UP
0001: wait 160
000A: 22@ += 1
00D6: if
0019:   22@ > 5
004D: jf @SHOW_SEL
0006: 22@ = 0
0002: jump @SHOW_SEL
:KEY_UP
00D6: if
0AB0:   key_pressed 38
004D: jf @PHONE_LOGIC2
0001: wait 160
000E: 22@ -= 1
00D6: if
001B:   0 > 22@
004D: jf @SHOW_SEL
0006: 22@ = 5

:SHOW_SEL
00D6: if
0039:   22@ == 0
004D: jf @SHOW_SEL2
0ACD: show_text_highpriority "CAMERA  Enter/Space snap" time 1200
0002: jump @PHONE_LOGIC2
:SHOW_SEL2
00D6: if
0039:   22@ == 1
004D: jf @SHOW_SEL3
0ACD: show_text_highpriority "INBOX" time 1200
0002: jump @PHONE_LOGIC2
:SHOW_SEL3
00D6: if
0039:   22@ == 2
004D: jf @SHOW_SEL4
0ACD: show_text_highpriority "CONTACTS" time 1200
0002: jump @PHONE_LOGIC2
:SHOW_SEL4
00D6: if
0039:   22@ == 3
004D: jf @SHOW_SEL5
0ACD: show_text_highpriority "STATUS" time 1200
0002: jump @PHONE_LOGIC2
:SHOW_SEL5
00D6: if
0039:   22@ == 4
004D: jf @SHOW_SEL6
0ACD: show_text_highpriority "HELP" time 1200
0002: jump @PHONE_LOGIC2
:SHOW_SEL6
0ACD: show_text_highpriority "CLOSE" time 1200

:PHONE_LOGIC2
// Enter (13) or Space (32) to activate
00D6: if or
0AB0:   key_pressed 13
0AB0:   key_pressed 32
004D: jf @KEY_BACK
0001: wait 180
00D6: if
0039:   22@ == 0
004D: jf @SEL_INBOX
0A2F: set_photo_camera_effect 1
0001: wait 80
0A1E: take_photo 1
0001: wait 200
0A2F: set_photo_camera_effect 0
018C: play_sound 1132 at 0.0 0.0 0.0
00D6: if
0AAB:   file_exists "cleo\GroveLink\link.ini"
004D: jf @SNAP_NOINI
0AF1: write_int 1 to_ini_file "cleo\GroveLink\link.ini" section "PHOTO" key "take"
0AF0: 25@ = read_int_from_ini_file "cleo\GroveLink\link.ini" section "PHOTO" key "count"
000A: 25@ += 1
0AF1: write_int 25@ to_ini_file "cleo\GroveLink\link.ini" section "PHOTO" key "count"
0ACE: show_formatted_text_highpriority "PHOTO TAKEN  #%d" time 2200 25@
0002: jump @KEY_BACK
:SNAP_NOINI
0ACD: show_text_highpriority "PHOTO TAKEN" time 2000
0002: jump @KEY_BACK

:SEL_INBOX
00D6: if
0039:   22@ == 1
004D: jf @SEL_CONTACTS
// INBOX: read msg from link.ini; clear new flag if set; sound when new
00D6: if
0AAB:   file_exists "cleo\GroveLink\link.ini"
004D: jf @INBOX_EMPTY
0AF0: 24@ = read_int_from_ini_file "cleo\GroveLink\link.ini" section "INBOX" key "new"
0AF4: 0@s = read_string_from_ini_file "cleo\GroveLink\link.ini" section "INBOX" key "msg"
00D6: if
0039:   24@ == 1
004D: jf @INBOX_SHOW
018C: play_sound 1052 at 0.0 0.0 0.0
0AF1: write_int 0 to_ini_file "cleo\GroveLink\link.ini" section "INBOX" key "new"
:INBOX_SHOW
0ACD: show_text_highpriority 0@s time 3500
0002: jump @KEY_BACK
:INBOX_EMPTY
0ACD: show_text_highpriority "INBOX EMPTY  start bridge" time 2000
0002: jump @KEY_BACK

:SEL_CONTACTS
00D6: if
0039:   22@ == 2
004D: jf @SEL_STATUS
// Flavor contacts — static 0ACD lines only, no ped models
00D6: if
0039:   26@ == 0
004D: jf @CONTACT1
0ACD: show_text_highpriority "SWEET: Stay Grove, cj" time 2800
0006: 26@ = 1
0002: jump @KEY_BACK
:CONTACT1
00D6: if
0039:   26@ == 1
004D: jf @CONTACT2
0ACD: show_text_highpriority "SMOKE: You need wheels?" time 2800
0006: 26@ = 2
0002: jump @KEY_BACK
:CONTACT2
00D6: if
0039:   26@ == 2
004D: jf @CONTACT3
0ACD: show_text_highpriority "RYDER: Homie check in" time 2800
0006: 26@ = 3
0002: jump @KEY_BACK
:CONTACT3
0ACD: show_text_highpriority "CESAR: Vatos forever" time 2800
0006: 26@ = 0
0002: jump @KEY_BACK

:SEL_STATUS
00D6: if
0039:   22@ == 3
004D: jf @SEL_HELP
00D6: if
0AAB:   file_exists "cleo\GroveLink\link.ini"
004D: jf @STATUS_NOINI
0AF0: 24@ = read_int_from_ini_file "cleo\GroveLink\link.ini" section "STATUS" key "bridge"
0AF0: 25@ = read_int_from_ini_file "cleo\GroveLink\link.ini" section "PHOTO" key "count"
00D6: if
0039:   24@ == 1
004D: jf @STATUS_DOWN
0ACE: show_formatted_text_highpriority "BRIDGE LIVE  %d shots" time 2800 25@
0002: jump @KEY_BACK
:STATUS_DOWN
0ACE: show_formatted_text_highpriority "NO BRIDGE  %d shots" time 2800 25@
0002: jump @KEY_BACK
:STATUS_NOINI
0ACD: show_text_highpriority "NO link.ini  run INSTALL" time 2200
0002: jump @KEY_BACK

:SEL_HELP
00D6: if
0039:   22@ == 4
004D: jf @SEL_CLOSE
// HELP: short 0ACD tips — K, Camera, START bridge, UPDATE if outdated
0ACD: show_text_highpriority "HELP: K opens phone" time 1800
0001: wait 1900
0ACD: show_text_highpriority "Camera: Enter or Space" time 1800
0001: wait 1900
0ACD: show_text_highpriority "OPEN START GROVELINK ON PC" time 2200
0001: wait 2300
0ACD: show_text_highpriority "Outdated? UPDATE_GROVELINK" time 2500
0002: jump @KEY_BACK

:SEL_CLOSE
0006: 20@ = 0
0ACD: show_text_highpriority "GROVELINK CLOSED" time 1200

:KEY_BACK
00D6: if
0AB0:   key_pressed 8
004D: jf @MAIN
0006: 20@ = 0
0006: 27@ = 0
0001: wait 160
0002: jump @MAIN
